7   package com.googlecode.catchexception.throwable;
8   
9   import java.io.Serializable;
10  
11  /**
12   * Thrown if a method has not thrown an throwable of the expected type.
13   */
14  public class ThrowableNotThrownAssertionError extends AssertionError {
15  
16      /**
17       * See {@link Serializable}.
18       */
19      private static final long serialVersionUID = 7044423604241785057L;
20  
21      /**
22       * Use this constructor if neither an throwable of the expected type nor another throwable is thrown.
23       *
24       * @param <E>
25       *            the type of the throwable that is not thrown.
26       * @param clazz
27       *            the type of the throwable that is not thrown.
28       */
29      public <E extends Throwable> ThrowableNotThrownAssertionError(Class<E> clazz) {
30          super(clazz == Throwable.class ? "Throwable expected but not thrown"
31                  : "Neither a throwable of type " + clazz.getName() + " nor another throwable was thrown");
32      }
33  
34      /**
35       * Use this constructor if an throwable of another than the expected type is thrown.
36       *
37       * @param <E>
38       *            the type of the throwable that is not thrown.
39       * @param clazz
40       *            the type of the throwable that is not thrown.
41       * @param e
42       *            the throwable that has been thrown instead of the expected one.
43       */
44      public <E extends Throwable> ThrowableNotThrownAssertionError(Class<E> clazz, Throwable e) {
45          super("Throwable of type " + clazz.getName() + " expected but was not thrown. " + "Instead a throwable of type "
46                  + e.getClass() + " with message '" + e.getMessage() + "' was thrown.");
47      }
48  
49  }
